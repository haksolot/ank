#!/usr/bin/env bash
# One line per task closed by a PR merged into main in the window, every date in UTC:
# 1 pr  2 merge  3 base  4 base_date  5 first_commit  6 earliest_trace  7 done  8 own_tip  9 id  10 pr_created  11 scope
# Only PRs whose merge commit is on main's first-parent chain: #264's merge reached main inside #263 and #265.
export TZ=UTC; main=98999e3635f71d46abc84af0b0d566233f995f81; fp=$(git rev-list --first-parent "$main")
done_ids() { git grep -l '^status: done$' "$1" -- '.ank/**/TASK-*.md' ':!.ank/log' | sed 's#.*/##; s#\.md$##' | sort; }
gh pr list --repo haksolot/ank --state merged --base main --search "merged:2026-08-01T00:00:00Z..2026-09-15T09:35:23Z" --limit 1000 \
  --json number,mergeCommit,createdAt --jq '.[] | "\(.number) \(.mergeCommit.oid) \(.createdAt)"' |
while read -r n m created; do
  grep -qx "$m" <<<"$fp" && git rev-parse -q --verify "$m^2" >/dev/null || continue
  ids=$(comm -13 <(done_ids "$m^1") <(done_ids "$m^2"))
  [ -n "$ids" ] || continue
  f=$(git rev-list --first-parent --reverse "$m^1..$m^2" | head -1)
  b=$(git rev-parse "$f^1")
  sync=$(git rev-list --first-parent --merges --reverse "$m^1..$m^2" | head -1)
  tip=$(if [ -n "$sync" ]; then git rev-parse "$sync^1"; else git rev-parse "$m^2"; fi)
  fc=$(git show -s --date=format-local:%Y-%m-%dT%H:%M:%SZ --format=%ad "$f")
  for id in $ids; do
    p=$(git ls-tree -r --name-only "$m^2" -- .ank | grep -v '^.ank/log/' | grep "/$id\.md$")
    dd=$(git log --reverse --date=format-local:%Y-%m-%dT%H:%M:%SZ --format=%ad -G'^status: done$' "$m^1..$m^2" -- "$p" | head -1)
    trace=$( { echo "$fc"; git diff --name-only --diff-filter=A "$m^1...$m^2" -- '.ank/**/LOG-*.md' |
      while read -r lp; do git show "$m^2:$lp" | awk -v id="$id" '/^created:/{c=$2} /^about:/{a=$2} END{if(a==id)print c}'; done; } | sort | head -1)
    scope=$(git show "$m^2:$p" | awk '/^scope:/{s=1;next} s&&/^  - /{sub(/^  - /,"");gsub(/"/,"");printf "%s,",$0;next} s{exit}')
    echo "$n $m $b $(git show -s --date=format-local:%Y-%m-%dT%H:%M:%SZ --format=%cd "$b") $fc $trace $dd $tip $id $created ${scope%,}"
  done
done

#!/usr/bin/env bash
# (a) per task PR: how many first-parent commits of main touching .ank/, made at or before the moment in
# column $1 of tasks.txt, are not reachable from the branch's base. Behind means at least one.
# 5 = first own commit, 6 = earliest trace (earliest over the PR's tasks), 10 = PR created
col=$1; main=98999e3635f71d46abc84af0b0d566233f995f81
sort -k1,1n -k"$col","$col" tasks.txt | awk '!seen[$1]++' |
while read -r line; do
  set -- $line; t=$(awk -v c="$col" '{print $c}' <<<"$line")
  echo "$1 $(git rev-list --first-parent --count --until="$t" "$main" "^$3" -- .ank/)"
done
