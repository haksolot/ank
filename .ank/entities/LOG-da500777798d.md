Rules. A pair is two tasks closed by two different PRs (tasks closed together in one PR share a branch and cannot conflict). A claim's window runs from its start to the author date of the commit that set the task done (tasks.txt column 7), because a claim ends at done. The start is not recorded anywhere any more, so two windows are counted: "trace", the earliest trace of the agent (column 6; late, since a claim precedes its first commit), and "base", the committer date of the commit the branch was cut from (column 4; early, since the branch cannot predate its base). Two windows overlap when each starts strictly before the other ends. Scopes intersect when glob_overlap, ported from crates/ank-cli/src/claim.rs and computed on each task's scope as it stood on its branch at merge (git show M^2:<entity>), names anything in common: the same computation claim prints. A git conflict is git merge-tree --write-tree --name-only --no-messages <tipA> <tipB> exiting 1 on a path both PRs changed (git diff --name-only M^1...M^2), where a tip is the branch before the first merge of main into it (tasks.txt column 8), so a conflict already resolved on the branch is still seen. A red first CI run is attempt 1 of the earliest ci.yml run on the later-merged PR's branch (firstrun.py) concluding failure.

--- pairs2.py
#!/usr/bin/env python3
# (b) python3 pairs2.py {trace|base}   reads tasks.txt, runs1.txt, heads.txt; runs git in the repository.
import sys, subprocess, re, itertools, collections
WIN = sys.argv[1]
git = lambda *a: subprocess.run(['git', *a], capture_output=True, text=True)

# glob_overlap, ported from crates/ank-cli/src/claim.rs (globset defaults: * crosses '/')
W = set('*?[{')
pat = lambda g: any(c in W for c in g)
def ldir(g):
    i = min([k for k, c in enumerate(g) if c in W], default=len(g)); j = g[:i].rfind('/')
    return g[:j] if j >= 0 else ''
under = lambda c, p: p == '' or c == p or c.startswith(p + '/')
def narrower(a, b):
    k = lambda g: (len(ldir(g)), sum(c not in W for c in g))
    return b if k(a) < k(b) else a if k(a) > k(b) else min(a, b)
def rx(g):
    o, i = '', 0
    while i < len(g):
        if g.startswith('**/', i): o += '(?:.*/)?'; i += 3; continue
        if g.startswith('**', i): o += '.*'; i += 2; continue
        c = g[i]
        if c == '*': o += '.*'
        elif c == '?': o += '.'
        elif c == '{': j = g.index('}', i); o += '(?:' + '|'.join(map(re.escape, g[i+1:j].split(','))) + ')'; i = j
        elif c == '[': j = g.index(']', i + 1); o += '[' + g[i+1:j] + ']'; i = j
        else: o += re.escape(c)
        i += 1
    return re.compile('^' + o + '$')
def overlap(a, b):
    a, b = a.rstrip('/'), b.rstrip('/')
    if a == b: return a
    if not pat(a) and not pat(b): return b if under(b, a) else a if under(a, b) else None
    if not pat(a) or not pat(b):
        path, p = (a, b) if not pat(a) else (b, a)
        return path if rx(p).match(path) else (p if under(ldir(p), path) else None)
    x, y = under(ldir(b), ldir(a)), under(ldir(a), ldir(b))
    return narrower(a, b) if x and y else b if x else a if y else None
common = lambda A, B: sorted({c for a in A for b in B for c in [overlap(a, b)] if c})

T = []
for line in open('tasks.txt'):
    f = line.split()
    T.append(dict(pr=int(f[0]), merge=f[1], start=f[5] if WIN == 'trace' else f[3], end=f[6], tip=f[7], id=f[8], scope=f[10].split(',')))
heads = {int(n): (h, m) for n, h, m in (l.split() for l in open('heads.txt'))}
runs = [l.split() for l in open('runs1.txt')]  # id attempt event branch sha created conclusion
prev, last = {}, {}
for n, (h, m) in sorted(heads.items(), key=lambda kv: kv[1][1]):
    prev[n] = last.get(h); last[h] = m
def first_red(n):
    h, m = heads[n]
    c = [r for r in runs if r[3] == h and r[5] <= m and (prev[n] is None or r[5] > prev[n])]
    return bool(c) and min(c, key=lambda r: (r[5], int(r[0])))[6] == 'failure'
files = {}
def pr_files(t):
    if t['pr'] not in files:
        files[t['pr']] = set(git('diff', '--name-only', f"{t['merge']}^1...{t['merge']}^2").stdout.split())
    return files[t['pr']]
def conflicted(a, b):
    r = git('merge-tree', '--write-tree', '--name-only', '--no-messages', a['tip'], b['tip'])
    return set(r.stdout.split()[1:]) if r.returncode == 1 else set()

P = []
for a, b in itertools.combinations(T, 2):
    if a['pr'] == b['pr'] or not (a['start'] < b['end'] and b['start'] < a['end']):
        continue
    first, later = sorted((a, b), key=lambda t: heads[t['pr']][1])
    shared = pr_files(a) & pr_files(b)
    P.append(dict(pair=f"{first['pr']}/{later['pr']} {first['id']} {later['id']}", common=common(a['scope'], b['scope']),
                  shared=sorted(shared), conflict=sorted(conflicted(a, b) & shared) if shared else [],
                  red=first_red(later['pr'])))
def line(label, ps):
    k = collections.Counter((bool(p['conflict']), p['red']) for p in ps)
    print(f"{label}: {len(ps)} pairs, a file edited by both {sum(bool(p['shared']) for p in ps)}, "
          f"git conflict {k[(True, False)] + k[(True, True)]}, red first run {k[(False, True)] + k[(True, True)]} "
          f"(both {k[(True, True)]}), neither {k[(False, False)]}")
print(f"window={WIN} tasks={len(T)} prs={len(set(t['pr'] for t in T))}")
line('claims overlapping in time', P)
line('  scopes intersect', [p for p in P if p['common']])
line('  scopes disjoint', [p for p in P if not p['common']])
if '-v' in sys.argv:
    for p in P:
        if p['common'] or p['conflict'] or p['red']:
            print(' ', p['pair'], 'common=' + ','.join(p['common']) if p['common'] else 'disjoint',
                  'conflict=' + ','.join(p['conflict']) if p['conflict'] else '', 'RED' if p['red'] else '')
